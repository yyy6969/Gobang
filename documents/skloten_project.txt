Gomoku/
├── core/                         # 游戏核心（与界面无关）

│   ├── player.h                  # Player 抽象接口

│   ├── ai_player.h/cpp           # AI 玩家（可暂简单实现）
│   └── timer_manager.h/cpp       # 双方计时器管理
├── network/                      # 网络对战

│   ├── game_server.h/cpp         # 局域网服务端
│   └── game_client.h/cpp         # 局域网客户端
├── ui/                           # QML 界面与 C++ 桥接
│   ├── qml/
│   │   ├── MainScreen.qml        # 主菜单（三个入口）

│   ├── game_controller.h/cpp     # 对局控制器（连接引擎、玩家与UI）
│   └── mainwindow.h/cpp          # 主窗口（装载 QML）
